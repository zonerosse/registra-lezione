cmake_minimum_required(VERSION 3.22)
project(vmwhisper C CXX)
# whisper.cpp (scaricato dal workflow in questa cartella) + il ponte vmwhisper.cpp, per l'iPhone
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(whisper.cpp)
add_library(vmwhisper SHARED vmwhisper.cpp)
target_link_libraries(vmwhisper whisper "-framework Accelerate" "-framework Foundation")
set_target_properties(vmwhisper PROPERTIES CXX_STANDARD 17)
